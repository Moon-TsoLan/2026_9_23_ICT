def complete_json(
    llm: LLMClient,
    *,
    step: str,
    prompt_version: str,
    user: str,
    validate,
    counter: Counter | None = None,
    thinking: bool = False,
    notes: dict | None = None,
    receiver=None,
) -> dict:
    """Repair representation/plan errors once; a receiver may retain independent valid items.

    Dependent merge plans never use a receiver. Business uncertainty must be returned as audit,
    not represented by a validation exception. An unexpected validator exception is an
    implementation error and is never sent back to the model as its own mistake.
    """
    payload = user
    attempts, issues = [], []
    for attempt in range(2):
        try:
            result = llm.complete(step=step, prompt_version=prompt_version, user=payload,
                                  thinking=thinking)
        except LLMError as exc:
            if receiver is None or not receiver.received:
                raise
            attempts.append({"attempt": attempt + 1, "issues": [ValidationIssue(str(exc), "transport").as_dict()]})
            return receiver.result(attempts)
        if notes is not None:
            notes["reasoning"] = (getattr(result, "reasoning", "") or "")[:1200]
            notes.setdefault("calls", []).append({"attempt": attempt + 1,
                "latency_ms": result.latency_ms, "usage": getattr(result, "usage", {}),
                "prompt_version": prompt_version, "model_name": result.model_name})
        if counter is not None:
            bump_count(counter, step)
        try:
            parsed = parse_json_object(result.text)
        except (ValueError, json.JSONDecodeError) as exc:
            issues = [ValidationIssue(str(exc))]
        else:
            try:
                checked = validate(parsed)
            except Exception as exc:
                issues = [ValidationIssue(str(exc), "implementation")]
                raise LLMValidationError(issues, attempts) from exc
            issues = ([checked] if isinstance(checked, ValidationIssue) else
                      [ValidationIssue(checked)] if checked else [])
            if not issues and receiver is not None:
                issues = receiver.accept(parsed, attempt)
            if not issues:
                attempts.append({"attempt": attempt + 1, "issues": []})
                if notes is not None:
                    notes["attempts"] = attempts
                return receiver.result(attempts) if receiver is not None else parsed
        attempts.append({"attempt": attempt + 1, "issues": [item.as_dict() for item in issues]})
        if notes is not None:
            notes["attempts"] = attempts
        if any(item.category not in {"schema", "plan_constraint"} for item in issues):
            raise LLMValidationError(issues, attempts)
        if attempt == 0:
            repair = {"previous_response": result.text,
                      "validation_issues": [item.as_dict() for item in issues]}
            if receiver is not None:
                repair["accepted_independent_outputs"] = receiver.result(attempts)
            payload = (user + "\n\n协议修复：以下是上次回答与具体错误。保留合法信息，只修复列出的错误；"
                       "带依赖的计划须返回完整合法计划。独立通道已接收的条目由程序保留，不必重复。\n" +
                       json.dumps(repair, ensure_ascii=False))
    if receiver is not None and receiver.received:
        return receiver.result(attempts)
    raise LLMValidationError(issues, attempts)
```

def align_fields(entity_type: str, raw_fields: dict[str, Any]) -> dict[str, Any]:
    """Map common wrong field names onto the contract keys. Known keys win."""
    aligned = dict(raw_fields)
    aliases = FIELD_ALIASES.get(entity_type, {})
    for source, target in aliases.items():
        if source not in raw_fields or target in aligned and aligned[target] not in (None, ""):
            continue
        aligned[target] = raw_fields[source]
    brand_model = raw_fields.get("brand_model") or raw_fields.get("品牌/型号")
    if brand_model not in (None, ""):
        if aligned.get("brand") in (None, ""):
            aligned["brand"] = brand_model
        if aligned.get("spec_model") in (None, "") and any(character.isdigit() for character in str(brand_model)):
            aligned["spec_model"] = brand_model
    return aligned
```

def observe(raw: Any, confidence: float | None = None) -> FieldObservation:
    if raw is None:
        return FieldObservation(raw_value=None, normalized_value=None, status="missing")
    text = raw if not isinstance(raw, str) else raw.strip()
    if text == "" or text in {"无", "-", "/"}:
        return FieldObservation(raw_value=None, normalized_value=None, status="missing", confidence=confidence)
    if isinstance(text, str) and any(token in text for token in POINTING) and len(text) < 12:
        return FieldObservation(
            raw_value=text, normalized_value=None, status="points_to_attachment", confidence=confidence
        )
    return FieldObservation(raw_value=text, normalized_value=None, status="present", confidence=confidence)
```

    def accept(self, parsed, attempt):
        self.received = True
        issues = []
        for channel in self.outputs:
            rows = parsed.get(channel, [])
            if not isinstance(rows, list):
                issue = ValidationIssue("需要数组", path=channel)
                issues.append(issue)
                self.audit.append({"attempt": attempt + 1, **issue.as_dict(), "raw": deepcopy(rows)})
                continue
            # Deduplicate re-emission across repair attempts, not independent same-response rows.
            previous = set(self.seen[channel])
            for index, row in enumerate(rows):
                error = self.check(channel, row)
                if error:
                    issue = ValidationIssue(error, path=f"{channel}[{index}]")
                    issues.append(issue)
                    self.audit.append({"attempt": attempt + 1, **issue.as_dict(), "raw": deepcopy(row)})
                    continue
                fingerprint = json.dumps(row, sort_keys=True, ensure_ascii=False)
                if fingerprint not in previous:
                    self.outputs[channel].append(deepcopy(row))
                self.seen[channel].add(fingerprint)
        return issues
```

def apply_native_resolutions(plans, decisions, candidates, resolutions, *, update_files=True):
    """Found evidence is not final resolution: require extracted values and matching ownership."""
    needs = {need.get("need_id"): need for plan in plans.projects for need in plan.needs}
    resolved = set()
    coverage = {}
    for resolution in resolutions:
        need = needs.get(resolution.get("need_id"))
        if need is None or need.get("purpose") != "fill_field" or need["field"] != resolution.get("field"):
            continue
        target = resolution.get("target_candidate_id")
        if target not in need.get("target_candidate_ids", []):
            continue
        sources = [item for item in candidates if item.evidence and
                       resolution.get("source_ref") in item.evidence.source_refs and
                       item.package_no == need["package_no"] and need["field"] in item.fields and
                       item.fields[need["field"]].status == "present"]
        # SUB resolutions must identify a unique observation, not any supplier on the same page.
        if need.get("target_entity") == "sub":
            sources = [item for item in sources if resolution.get("source_candidate_ref") in item.evidence.source_refs
                       and item.entity_type == "sub"
                       and (item.fields[need["field"]].normalized_value if item.fields[need["field"]].normalized_value is not None
                            else item.fields[need["field"]].raw_value) not in (None, "", "unknown")]
        source = sources[0] if len(sources) == 1 else None
        if source is not None:
            coverage.setdefault(need["need_id"], {})[target] = resolution["source_ref"]
    for need_id, targets in coverage.items():
        need = needs[need_id]
        need["source_refs"] = list(dict.fromkeys(targets.values()))
        need["covered_target_candidate_ids"] = list(targets)
        if set(need["target_candidate_ids"]) <= set(targets):
            need["state"] = "evidence_ready"
            resolved.add(need_id)
    if not update_files:
        return resolved
    for decision in decisions.file_decisions:
        if not decision.evidence_refs:
            continue
        packages = {item.package_no for item in candidates if item.evidence and
                    set(decision.evidence_refs) & set(item.evidence.source_refs) and item.package_no is not None}
        hit_ids = getattr(decision, "needs_hit_ids", [])
        relevant = [need for need in needs.values() if need["field"] in decision.expected_fields
                    and need.get("purpose") != "discover_objects" and need.get("package_no") in packages
                    and (not hit_ids or need.get("need_id") in hit_ids)]
        # A discovery or binding task cannot be declared done by an ordinary field observation.
        complete = bool(relevant) and all(need.get("need_id") in resolved for need in relevant)
        if complete and not (getattr(decision, "screen", None) or {}).get("requires_deep_parse", True):
            decision.read_strategy = "skip"
            decision.reuse_status = "native_evidence_ready"
            # Discovery stays pending. This file was judged by the model not to need deeper parsing;
            # that does not certify the announcement's complete object inventory.
            decision.reason += ":native_evidence_ready"
        else:
            decision.reuse_status = "native_evidence_partial"
    return resolved

def _merge_entity(items: list[Candidate], entity_type: str):
    chosen: dict[tuple, Candidate] = {}
    order: list[tuple] = []
    conflicts: list[dict] = []
    unmatched: list[dict] = []
    keys = SUB_FIELDS
    # Preserve all new observations before the old fallback mutates its selected candidate.
    observed = {}
    for item in items:
        if item.entity_type != entity_type:
            continue
        identity = (MEMBER_RE.sub("", _norm_name(_value(item, "supplier_name"))),)
        values = observed.setdefault(identity, {field: [] for field in SUB_FIELDS[4:]})
        for field in SUB_FIELDS[4:]:
            value = _value(item, field)
            if value not in (None, "", "unknown") and value not in values[field]:
                values[field].append(value)
    for candidate in items:
        if candidate.entity_type != entity_type:
            continue
        key = (MEMBER_RE.sub("", _norm_name(_value(candidate, "supplier_name"))),)
        if not key[0]:
            continue
        if key not in chosen:
            chosen[key] = candidate
            order.append(key)
            continue
        current = chosen[key]
        replace = candidate.source_priority > current.source_priority or (
            candidate.source_priority == current.source_priority and _completeness(candidate) > _completeness(current)
        )
        base, extra = (candidate, current) if replace else (current, candidate)
        html_names = [item for item in (current, candidate) if item.source.source_type == "html"]
        if html_names:
            base.fields["supplier_name"].normalized_value = _value(html_names[0], "supplier_name")
        for field_key in keys:
            if field_key == "supplier_name":
                continue
            left = _value(base, field_key)
            right = _value(extra, field_key)
            if left in (None, "") and right not in (None, ""):
                base.fields[field_key].normalized_value = right
                base.fields[field_key].status = "present"
            elif field_key == "is_winner" and left in (None, False, "") and right in (True, "是", "true"):
                base.fields[field_key].normalized_value = right
                base.fields[field_key].status = "present"
            elif left not in (None, "") and right not in (None, "") and not _same(left, right) and field_key != "is_winner":
                base.fields[field_key].status = "conflict"
                conflicts.append(
                    {"field": field_key, "candidate_ids": [base.candidate_id, extra.candidate_id], "values": [left, right]}
                )
        chosen[key] = base
    merged = []
    ids = []
    for key in order:
        candidate = chosen[key]
        ids.append(candidate.candidate_id)
        if not _value(candidate, "supplier_name"):
            continue
        winner = _value(candidate, "is_winner")
        new_values = {}
        for field, values in observed[key].items():
            new_values[field] = values[0] if len(values) == 1 else None
            if len(values) > 1:
                conflicts.append({"field": field, "supplier_name": _value(candidate, "supplier_name"),
                                  "values": values, "reason": "sub_fallback_new_field_unresolved"})
        new_values["review_status"] = new_values["review_status"] or "unknown"
        if new_values["review_status"] != "failed":
            new_values["review_failure_reason"] = None
        if new_values["review_status"] == "failed" and winner:
            conflicts.append({"field": "is_winner", "supplier_name": _value(candidate, "supplier_name"),
                              "reason": "winner_review_conflict"})
            winner = False
        merged.append(
            Sub(
                supplier_name=str(_value(candidate, "supplier_name")),
                score=_value(candidate, "score"),
                is_winner=bool(winner) and str(winner).lower() not in {"false", "否", "0"},
                supplier_award_amount=_value(candidate, "supplier_award_amount"),
                **new_values,
            )
        )
    return merged, ids, conflicts, unmatched
